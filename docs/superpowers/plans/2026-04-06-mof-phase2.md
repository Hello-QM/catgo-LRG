```rust
//! Functional group identification on MOF linkers/ligands.
//!
//! A functional group consists of non-C/H atoms on a Linker or Ligand SBU
//! that do NOT coordinate to any metal node, plus their attached H atoms.
//! Examples: -NH2, -OH, -NO2, -F, -Br, -SH, -SO3H.

use std::collections::{HashSet, VecDeque};
use serde::{Deserialize, Serialize};
use crate::element::Element;
use super::periodic_graph::PeriodicGraph;
use super::{MofClusters, SbuType};

/// A detected functional group on a linker/ligand.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FunctionalGroup {
    /// Atom indices belonging to this functional group.
    pub atom_indices: Vec<usize>,
    /// Chemical name (e.g., "NH2", "OH", "NO2", "F").
    pub name: String,
    /// Index of the parent SBU (linker or ligand).
    pub parent_sbu: usize,
    /// Index of the atom connecting this group to the organic backbone.
    pub attachment_atom: usize,
}

/// Detect functional groups on Linker and Ligand SBUs.
///
/// Algorithm:
/// 1. For each Linker/Ligand SBU, find non-C/H atoms that don't bond to any Node.
/// 2. BFS from each such atom collecting connected non-C atoms + H neighbors.
/// 3. Name the group from its element composition.
pub fn detect_functional_groups(
    clusters: &MofClusters,
    graph: &PeriodicGraph,
    elements: &[Element],
) -> Vec<FunctionalGroup> {
    let mut groups = Vec::new();
    let n = elements.len();

    // Build set of atoms in Node SBUs (for "coordinates metal" check)
    let node_atoms: HashSet<usize> = clusters.sbus.iter()
        .filter(|s| s.sbu_type == SbuType::Node)
        .flat_map(|s| s.atom_indices.iter().copied())
        .collect();

    for (sbu_idx, sbu) in clusters.sbus.iter().enumerate() {
        if sbu.sbu_type != SbuType::Linker && sbu.sbu_type != SbuType::Ligand {
            continue;
        }

        let sbu_set: HashSet<usize> = sbu.atom_indices.iter().copied().collect();
        let mut visited: HashSet<usize> = HashSet::new();

        for &atom in &sbu.atom_indices {
            if atom >= n || visited.contains(&atom) {
                continue;
            }
            let el = elements[atom];
            // Skip C and H — they're backbone
            if el == Element::C || el == Element::H {
                continue;
            }
            // Skip if this atom bonds to a Node atom (it's a coordination atom, not functional group)
            let bonds_to_node = graph.neighbors(atom).iter().any(|nbr| node_atoms.contains(&nbr.v));
            if bonds_to_node {
                continue;
            }

            // BFS to collect connected non-C/non-backbone atoms + their H
            let mut group_atoms = Vec::new();
            let mut queue = VecDeque::new();
            queue.push_back(atom);
            visited.insert(atom);
            let mut attachment = atom; // will be refined

            while let Some(u) = queue.pop_front() {
                group_atoms.push(u);
                for nbr in graph.neighbors(u) {
                    if nbr.v >= n || visited.contains(&nbr.v) || !sbu_set.contains(&nbr.v) {
                        continue;
                    }
                    let nbr_el = elements[nbr.v];
                    if nbr_el == Element::H {
                        // Always include H neighbors
                        visited.insert(nbr.v);
                        group_atoms.push(nbr.v);
                    } else if nbr_el != Element::C {
                        // Continue BFS through non-C atoms (e.g., -NO2, -SO3H)
                        let nbr_bonds_to_node = graph.neighbors(nbr.v).iter()
                            .any(|n2| node_atoms.contains(&n2.v));
                        if !nbr_bonds_to_node {
                            visited.insert(nbr.v);
                            queue.push_back(nbr.v);
                        }
                    }
                }
            }

            if group_atoms.is_empty() {
                continue;
            }

            // Find attachment atom: the C atom in the SBU bonded to this group
            for &g_atom in &group_atoms {
                for nbr in graph.neighbors(g_atom) {
                    if sbu_set.contains(&nbr.v) && elements[nbr.v] == Element::C {
                        attachment = nbr.v;
                        break;
                    }
                }
            }

            let name = name_functional_group(&group_atoms, elements);
            groups.push(FunctionalGroup {
                atom_indices: group_atoms,
                name,
                parent_sbu: sbu_idx,
                attachment_atom: attachment,
            });
        }
    }
    groups
}

/// Name a functional group from its element composition.
fn name_functional_group(atoms: &[usize], elements: &[Element]) -> String {
    let mut counts = std::collections::BTreeMap::new();
    for &idx in atoms {
        if idx < elements.len() {
            *counts.entry(elements[idx].symbol()).or_insert(0u32) += 1;
        }
    }

    // Try to match known patterns
    let key: Vec<(&str, u32)> = counts.iter().map(|(&s, &c)| (s, c)).collect();

    match key.as_slice() {
        [("F", 1)] => "F".to_string(),
        [("Cl", 1)] => "Cl".to_string(),
        [("Br", 1)] => "Br".to_string(),
        [("I", 1)] => "I".to_string(),
        [("H", 1), ("O", 1)] | [("O", 1), ("H", 1)] => "OH".to_string(),
        [("H", 1), ("S", 1)] | [("S", 1), ("H", 1)] => "SH".to_string(),
        [("H", 2), ("N", 1)] | [("N", 1), ("H", 2)] => "NH2".to_string(),
        [("N", 1), ("O", 2)] => "NO2".to_string(),
        [("H", 1), ("O", 3), ("S", 1)] | [("O", 3), ("S", 1), ("H", 1)] => "SO3H".to_string(),
        _ => {
            // Generic: concatenate element symbols with counts
            let mut parts = Vec::new();
            for (&sym, &c) in &counts {
                if c == 1 { parts.push(sym.to_string()); }
                else { parts.push(format!("{}{}", sym, c)); }
            }
            parts.join("")
        }
    }
}

//! 1D Rod SBU detection for MOF topology analysis.
//!
//! Rod SBUs are infinite metal chains periodic in exactly 1 dimension
//! (e.g., MIL-53 Al chains, MOF-74 metal helices).
//! These should NOT be split by resolve_periodic_sbus.

use std::collections::{HashMap, HashSet, VecDeque};
use super::periodic_graph::PeriodicGraph;
use super::classify::CLASS_INORGANIC;
use super::clustering::ClusteringState;
use super::SbuType;

/// Check if a set of atoms forms a 1D periodic chain.
///
/// Performs BFS tracking periodic offsets. If all offset vectors
/// are collinear (span a 1D subspace), it's a rod.
pub fn is_1d_periodic(
    graph: &PeriodicGraph,
    atoms: &[usize],
    classes: &[i32],
) -> bool {
    if atoms.is_empty() { return false; }
    let atom_set: HashSet<usize> = atoms.iter().copied().collect();
    let class = classes[atoms[0]];

    let mut offsets_seen: Vec<[i32; 3]> = Vec::new();
    let mut visited: HashMap<usize, [i32; 3]> = HashMap::new();
    let mut queue = VecDeque::new();

    visited.insert(atoms[0], [0, 0, 0]);
    queue.push_back((atoms[0], [0i32, 0, 0]));

    while let Some((u, u_ofs)) = queue.pop_front() {
        for nbr in graph.neighbors(u) {
            if !atom_set.contains(&nbr.v) || classes[nbr.v] != class {
                continue;
            }
            let v_ofs = [
                u_ofs[0] + nbr.ofs[0],
                u_ofs[1] + nbr.ofs[1],
                u_ofs[2] + nbr.ofs[2],
            ];
            if let Some(&existing) = visited.get(&nbr.v) {
                let diff = [v_ofs[0] - existing[0], v_ofs[1] - existing[1], v_ofs[2] - existing[2]];
                if diff != [0, 0, 0] {
                    offsets_seen.push(diff);
                }
            } else {
                visited.insert(nbr.v, v_ofs);
                queue.push_back((nbr.v, v_ofs));
            }
        }
    }

    is_1d_periodic_from_offsets(&offsets_seen)
}

/// Check if a set of offset vectors spans exactly 1 dimension (all collinear).
fn is_1d_periodic_from_offsets(offsets: &[[i32; 3]]) -> bool {
    if offsets.is_empty() { return false; }

    // Find first non-zero offset as reference direction
    let ref_dir = match offsets.iter().find(|o| **o != [0, 0, 0]) {
        Some(d) => *d,
        None => return false,
    };

    // All other offsets must be parallel to ref_dir (cross product = 0)
    offsets.iter().all(|o| {
        if *o == [0, 0, 0] { return true; }
        let cross = [
            ref_dir[1] * o[2] - ref_dir[2] * o[1],
            ref_dir[2] * o[0] - ref_dir[0] * o[2],
            ref_dir[0] * o[1] - ref_dir[1] * o[0],
        ];
        cross == [0, 0, 0]
    })
}

/// Detect rod SBUs and reclassify them from Node to Rod.
/// Must be called BEFORE resolve_periodic_sbus to prevent splitting.
pub fn detect_and_mark_rods(
    graph: &PeriodicGraph,
    state: &mut ClusteringState,
) {
    for sbu_idx in 0..state.sbus.len() {
        if state.sbus[sbu_idx].sbu_type != SbuType::Node || !state.sbus[sbu_idx].is_periodic {
            continue;
        }
        if is_1d_periodic(graph, &state.sbus[sbu_idx].atom_indices, &state.classes) {
            state.sbus[sbu_idx].sbu_type = SbuType::Rod;
        }
    }
}

## Task 3: RAC Descriptor Engine (Rust)

**Files:**

RAC (Revised Autocorrelation) descriptors: BFS over molecular graph collecting atomic property products/differences at each depth. Reference: molSimplify `graph_racs.py`.

Algorithm:
```
P_d(prop) = sum( prop(origin) * prop(j) )  for all j at graph distance d
D_d(prop) = sum( prop(origin) - prop(j) )  for all j at graph distance d
```
5 properties (Z, chi, T, I, S) x 4 scopes (mc, lc, func, f-sbu) x depths 0-3 x 2 ops (P, D) = ~160 features (some scopes may be empty).

//! Revised Autocorrelation (RAC) descriptor engine for MOFs.
//!
//! Computes graph-based autocorrelation functions following molSimplify.
//! Reference: Janet & Kulik, J. Phys. Chem. A 2017, 121, 8939.

use std::collections::{HashMap, VecDeque, HashSet};
use serde::{Deserialize, Serialize};
use crate::element::Element;
use super::periodic_graph::PeriodicGraph;
use super::{MofClusters, SbuType};

/// RAC atomic properties.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum RacProperty {
    Z,   // Nuclear charge (atomic number)
    Chi, // Pauling electronegativity
    T,   // Coordination number (graph degree)
    I,   // Identity (always 1)
    S,   // Covalent radius
}

/// RAC operation type.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum RacOp {
    Product,    // P_d = sum( prop(origin) * prop(j) )
    Difference, // D_d = sum( prop(origin) - prop(j) )
}

/// RAC scope (which atoms to start BFS from).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum RacScope {
    MetalCentered, // mc: start from metal atoms in nodes
    LinkerCoord,   // lc: start from linker atoms coordinating to metals
    FuncGroup,     // func: start from functional group atoms
    FullSbu,       // f-sbu: start from all atoms in each SBU type
}

/// A single RAC descriptor value.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RacDescriptor {
    pub scope: RacScope,
    pub property: RacProperty,
    pub depth: usize,
    pub op: RacOp,
    pub value: f64,
    /// Human-readable name like "mc-Z-0-P"
    pub name: String,
}

/// Full RAC descriptor set for a MOF.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RacResult {
    pub descriptors: Vec<RacDescriptor>,
}

const MAX_DEPTH: usize = 3;
const PROPERTIES: [RacProperty; 5] = [
    RacProperty::Z, RacProperty::Chi, RacProperty::T,
    RacProperty::I, RacProperty::S,
];
const OPS: [RacOp; 2] = [RacOp::Product, RacOp::Difference];

/// Compute all RAC descriptors for a MOF structure.
pub fn compute_rac(
    clusters: &MofClusters,
    graph: &PeriodicGraph,
    elements: &[Element],
    functional_group_atoms: &[Vec<usize>],
) -> RacResult {
    let n = elements.len();

    // Build simple adjacency list (no periodic offsets needed for BFS depth)
    let adj = build_adjacency(graph, n);

    // Compute property vectors
    let props = build_property_maps(elements, &adj);

    // Determine starting atoms for each scope
    let mc_atoms = get_metal_centered_atoms(clusters, elements);
    let lc_atoms = get_linker_coordinating_atoms(clusters, graph, elements);
    let func_atoms: Vec<usize> = functional_group_atoms.iter().flatten().copied().collect();

    let mut descriptors = Vec::new();

    let scopes = [
        (RacScope::MetalCentered, &mc_atoms),
        (RacScope::LinkerCoord, &lc_atoms),
        (RacScope::FuncGroup, &func_atoms),
    ];

    for (scope, starting_atoms) in &scopes {
        if starting_atoms.is_empty() { continue; }
        let scope_descs = compute_rac_scope(&adj, &props, MAX_DEPTH, starting_atoms);
        for ((prop, depth, op), value) in scope_descs {
            let name = format!("{}-{}-{}-{}",
                scope_name(*scope), prop_name(prop), depth, op_name(op));
            descriptors.push(RacDescriptor {
                scope: *scope, property: prop, depth, op, value, name,
            });
        }
    }

    // f-sbu scope: compute separately for node and linker SBU atoms
    for sbu_label in &["node", "linker"] {
        let sbu_type = if *sbu_label == "node" { SbuType::Node } else { SbuType::Linker };
        let atoms: Vec<usize> = clusters.sbus.iter()
            .filter(|s| s.sbu_type == sbu_type)
            .flat_map(|s| s.atom_indices.iter().copied())
            .collect();
        if atoms.is_empty() { continue; }
        let scope_descs = compute_rac_scope(&adj, &props, MAX_DEPTH, &atoms);
        for ((prop, depth, op), value) in scope_descs {
            let name = format!("f-{}-{}-{}-{}", sbu_label, prop_name(prop), depth, op_name(op));
            descriptors.push(RacDescriptor {
                scope: RacScope::FullSbu, property: prop, depth, op, value, name,
            });
        }
    }

    RacResult { descriptors }
}

/// Compute RAC for a single scope. Returns map of (property, depth, op) -> value.
pub fn compute_rac_scope(
    adj: &[Vec<usize>],
    props: &HashMap<RacProperty, Vec<f64>>,
    max_depth: usize,
    starting_atoms: &[usize],
) -> Vec<((RacProperty, usize, RacOp), f64)> {
    let mut results = Vec::new();
    for &prop in &PROPERTIES {
        let prop_vals = match props.get(&prop) {
            Some(v) => v,
            None => continue,
        };
        for depth in 0..=max_depth {
            for &op in &OPS {
                let val = compute_rac_single(adj, prop_vals, depth, op, starting_atoms);
                results.push(((prop, depth, op), val));
            }
        }
    }
    results
}

/// Compute a single RAC value: sum over all starting atoms of
/// the pairwise operation with atoms at exactly `depth` graph distance.
pub fn compute_rac_single(
    adj: &[Vec<usize>],
    prop_vals: &[f64],
    depth: usize,
    op: RacOp,
    starting_atoms: &[usize],
) -> f64 {
    let mut total = 0.0;
    for &origin in starting_atoms {
        if origin >= prop_vals.len() { continue; }
        let p_origin = prop_vals[origin];
        let atoms_at_depth = bfs_at_depth(adj, origin, depth);
        for target in atoms_at_depth {
            if target >= prop_vals.len() { continue; }
            let p_target = prop_vals[target];
            total += match op {
                RacOp::Product => p_origin * p_target,
                RacOp::Difference => p_origin - p_target,
            };
        }
    }
    total
}

/// BFS returning atoms at exactly `depth` from `start`.
fn bfs_at_depth(adj: &[Vec<usize>], start: usize, depth: usize) -> Vec<usize> {
    if depth == 0 { return vec![start]; }
    let mut visited = HashSet::new();
    let mut queue = VecDeque::new();
    visited.insert(start);
    queue.push_back((start, 0usize));
    let mut result = Vec::new();

    while let Some((u, d)) = queue.pop_front() {
        if d == depth {
            result.push(u);
            continue;
        }
        if d > depth { break; }
        for &v in &adj[u] {
            if !visited.contains(&v) {
                visited.insert(v);
                queue.push_back((v, d + 1));
            }
        }
    }
    result
}

fn build_adjacency(graph: &PeriodicGraph, n: usize) -> Vec<Vec<usize>> {
    let mut adj = vec![Vec::new(); n];
    for u in 0..n {
        for nbr in graph.neighbors(u) {
            if nbr.v < n && !adj[u].contains(&nbr.v) {
                adj[u].push(nbr.v);
            }
        }
    }
    adj
}

fn build_property_maps(elements: &[Element], adj: &[Vec<usize>]) -> HashMap<RacProperty, Vec<f64>> {
    let n = elements.len();
    let mut m = HashMap::new();
    m.insert(RacProperty::Z, elements.iter().map(|e| e.atomic_number() as f64).collect());
    m.insert(RacProperty::Chi, elements.iter().map(|e| e.electronegativity().unwrap_or(0.0)).collect());
    m.insert(RacProperty::T, (0..n).map(|i| adj[i].len() as f64).collect());
    m.insert(RacProperty::I, vec![1.0; n]);
    m.insert(RacProperty::S, elements.iter().map(|e| e.covalent_radius().unwrap_or(1.0)).collect());
    m
}

fn get_metal_centered_atoms(clusters: &MofClusters, elements: &[Element]) -> Vec<usize> {
    clusters.sbus.iter()
        .filter(|s| s.sbu_type == SbuType::Node || s.sbu_type == SbuType::Rod)
        .flat_map(|s| s.atom_indices.iter())
        .filter(|&&i| i < elements.len() && elements[i].is_metal())
        .copied()
        .collect()
}

fn get_linker_coordinating_atoms(
    clusters: &MofClusters, graph: &PeriodicGraph, elements: &[Element],
) -> Vec<usize> {
    let node_atoms: HashSet<usize> = clusters.sbus.iter()
        .filter(|s| s.sbu_type == SbuType::Node || s.sbu_type == SbuType::Rod)
        .flat_map(|s| s.atom_indices.iter().copied())
        .collect();

    clusters.sbus.iter()
        .filter(|s| s.sbu_type == SbuType::Linker)
        .flat_map(|s| s.atom_indices.iter())
        .filter(|&&atom| {
            atom < elements.len() &&
            graph.neighbors(atom).iter().any(|nbr| node_atoms.contains(&nbr.v))
        })
        .copied()
        .collect()
}

fn scope_name(s: RacScope) -> &'static str {
    match s {
        RacScope::MetalCentered => "mc",
        RacScope::LinkerCoord => "lc",
        RacScope::FuncGroup => "func",
        RacScope::FullSbu => "f",
    }
}

fn prop_name(p: RacProperty) -> &'static str {
    match p {
        RacProperty::Z => "Z",
        RacProperty::Chi => "chi",
        RacProperty::T => "T",
        RacProperty::I => "I",
        RacProperty::S => "S",
    }
}

fn op_name(o: RacOp) -> &'static str {
    match o {
        RacOp::Product => "P",
        RacOp::Difference => "D",
    }
}

//! Weisfeiler-Lehman graph hash for SBU/linker deduplication.
//!
//! Iteratively relabels nodes by hashing their label + sorted neighbor labels.
//! After `k` iterations, the sorted multiset of labels gives a canonical hash.

use std::collections::hash_map::DefaultHasher;
use std::hash::{Hash, Hasher};
use serde::{Deserialize, Serialize};
use crate::element::Element;
use super::periodic_graph::PeriodicGraph;
use super::{MofClusters, Sbu};

/// Result of WL hashing for an SBU.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct WlHashResult {
    pub sbu_index: usize,
    pub hash: u64,
}

/// Compute WL hash for a subgraph defined by node labels and adjacency.
pub fn wl_hash(labels: &[&str], adj: &[Vec<usize>], iterations: usize) -> u64 {
    let n = labels.len();
    let mut current_labels: Vec<u64> = labels.iter().map(|l| hash_str(l)).collect();

    for _ in 0..iterations {
        let mut new_labels = Vec::with_capacity(n);
        for i in 0..n {
            let mut nbr_labels: Vec<u64> = adj[i].iter()
                .filter(|&&j| j < n)
                .map(|&j| current_labels[j])
                .collect();
            nbr_labels.sort();

            let mut hasher = DefaultHasher::new();
            current_labels[i].hash(&mut hasher);
            nbr_labels.hash(&mut hasher);
            new_labels.push(hasher.finish());
        }
        current_labels = new_labels;
    }

    // Final hash: sorted multiset of all node labels
    current_labels.sort();
    let mut hasher = DefaultHasher::new();
    current_labels.hash(&mut hasher);
    hasher.finish()
}

/// Compute WL hashes for all SBUs in a MOF.
pub fn compute_sbu_hashes(
    clusters: &MofClusters,
    graph: &PeriodicGraph,
    elements: &[Element],
    iterations: usize,
) -> Vec<WlHashResult> {
    clusters.sbus.iter().enumerate().map(|(sbu_idx, sbu)| {
        let (labels, adj) = extract_sbu_subgraph(sbu, graph, elements);
        let label_refs: Vec<&str> = labels.iter().map(|s| s.as_str()).collect();
        let hash = wl_hash(&label_refs, &adj, iterations);
        WlHashResult { sbu_index: sbu_idx, hash }
    }).collect()
}

fn extract_sbu_subgraph(
    sbu: &Sbu,
    graph: &PeriodicGraph,
    elements: &[Element],
) -> (Vec<String>, Vec<Vec<usize>>) {
    let n = sbu.atom_indices.len();
    let idx_map: std::collections::HashMap<usize, usize> = sbu.atom_indices.iter()
        .enumerate()
        .map(|(local, &global)| (global, local))
        .collect();

    let labels: Vec<String> = sbu.atom_indices.iter().map(|&i| {
        if i < elements.len() { elements[i].symbol().to_string() }
        else { "X".to_string() }
    }).collect();

    let adj: Vec<Vec<usize>> = sbu.atom_indices.iter().map(|&global_idx| {
        graph.neighbors(global_idx).iter()
            .filter_map(|nbr| idx_map.get(&nbr.v).copied())
            .collect()
    }).collect();

    (labels, adj)
}

fn hash_str(s: &str) -> u64 {
    let mut hasher = DefaultHasher::new();
    s.hash(&mut hasher);
    hasher.finish()
}

//! Cap (ligand) replacement engine for MOF structures.
//!
//! Workflow:
//! 1. Find cap SBUs and their attachment points to nodes
//! 2. Delete all cap atoms
//! 3. Align new molecular fragment to each attachment point
//! 4. Insert fragment atoms into the structure

use nalgebra::Vector3;
use serde::{Deserialize, Serialize};
use crate::bonding::Bond;
use crate::element::Element;
use crate::species::{SiteOccupancy, Species};
use crate::structure::Structure;
use super::periodic_graph::PeriodicGraph;
use super::{MofClusters, SbuType};

/// A molecular fragment to insert at each cap attachment point.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MolecularFragment {
    /// Elements of each atom in the fragment.
    pub elements: Vec<Element>,
    /// Cartesian coordinates of each atom (Angstroms).
    pub cart_coords: Vec<Vector3<f64>>,
    /// Index of the atom that bonds to the node attachment point.
    pub bonding_atom_idx: usize,
}

/// Anchor information for a cap SBU.
#[derive(Debug, Clone)]
pub struct CapAnchor {
    /// Index of the cap (Ligand) SBU.
    pub cap_sbu_idx: usize,
    /// Atom on the node/PE side where the cap was bonded.
    pub node_attachment_atom: usize,
    /// Atom on the cap side that was bonded to the node (the anchor).
    pub cap_anchor_atom: usize,
    /// Bond direction from node_attachment → cap_anchor (Cartesian, normalized).
    pub bond_direction: Vector3<f64>,
    /// Bond image offset.
    pub bond_image: [i32; 3],
}

/// Find all cap attachment points in the MOF.
pub fn find_cap_anchors(
    clusters: &MofClusters,
    graph: &PeriodicGraph,
) -> Vec<CapAnchor> {
    let mut anchors = Vec::new();

    for (sbu_idx, sbu) in clusters.sbus.iter().enumerate() {
        if sbu.sbu_type != SbuType::Ligand {
            continue;
        }

        let cap_set: std::collections::HashSet<usize> =
            sbu.atom_indices.iter().copied().collect();

        // Find bonds from cap atoms to non-cap atoms (node/PE)
        for &atom in &sbu.atom_indices {
            for nbr in graph.neighbors(atom) {
                if !cap_set.contains(&nbr.v) {
                    let nbr_sbu = clusters.attributions[nbr.v];
                    if nbr_sbu < clusters.sbus.len() &&
                       (clusters.sbus[nbr_sbu].sbu_type == SbuType::Node ||
                        clusters.sbus[nbr_sbu].sbu_type == SbuType::PointOfExtension ||
                        clusters.sbus[nbr_sbu].sbu_type == SbuType::Rod)
                    {
                        anchors.push(CapAnchor {
                            cap_sbu_idx: sbu_idx,
                            node_attachment_atom: nbr.v,
                            cap_anchor_atom: atom,
                            bond_direction: Vector3::zeros(), // computed later with structure
                            bond_image: nbr.ofs,
                        });
                    }
                }
            }
        }
    }
    anchors
}

/// Replace all caps in a MOF with a new molecular fragment.
///
/// Returns a new Structure with caps removed and fragments inserted.
pub fn replace_caps(
    structure: &Structure,
    bonds: &[Bond],
    clusters: &MofClusters,
    fragment: &MolecularFragment,
) -> Result<Structure, String> {
    if fragment.bonding_atom_idx >= fragment.elements.len() {
        return Err("bonding_atom_idx out of range".to_string());
    }

    let graph = PeriodicGraph::from_bonds(structure.frac_coords.len(), bonds);
    let mut anchors = find_cap_anchors(clusters, &graph);

    if anchors.is_empty() {
        return Err("No cap SBUs found to replace".to_string());
    }

    // Compute bond directions in Cartesian space
    let lat_matrix = structure.lattice.matrix();
    for anchor in &mut anchors {
        let pos_node = lat_matrix * structure.frac_coords[anchor.node_attachment_atom];
        let img_offset = Vector3::new(
            anchor.bond_image[0] as f64,
            anchor.bond_image[1] as f64,
            anchor.bond_image[2] as f64,
        );
        let pos_cap = lat_matrix * (structure.frac_coords[anchor.cap_anchor_atom] + img_offset);
        let dir = pos_cap - pos_node;
        let len = dir.norm();
        anchor.bond_direction = if len > 1e-10 { dir / len } else { Vector3::x() };
    }

    // Collect atoms to remove (all cap SBU atoms)
    let cap_sbu_indices: std::collections::HashSet<usize> = anchors.iter()
        .map(|a| a.cap_sbu_idx)
        .collect();
    let atoms_to_remove: std::collections::HashSet<usize> = cap_sbu_indices.iter()
        .flat_map(|&sbu_idx| clusters.sbus[sbu_idx].atom_indices.iter().copied())
        .collect();

    // Build new structure: keep non-cap atoms, add fragment atoms at each anchor
    let mut new_occupancies = Vec::new();
    let mut new_frac_coords = Vec::new();

    // Keep non-cap atoms
    for (i, (occ, frac)) in structure.site_occupancies.iter()
        .zip(structure.frac_coords.iter())
        .enumerate()
    {
        if !atoms_to_remove.contains(&i) {
            new_occupancies.push(occ.clone());
            new_frac_coords.push(*frac);
        }
    }

    // Deduplicate anchors by cap SBU (one fragment per cap, not per anchor bond)
    let mut seen_caps = std::collections::HashSet::new();
    let unique_anchors: Vec<&CapAnchor> = anchors.iter()
        .filter(|a| seen_caps.insert(a.cap_sbu_idx))
        .collect();

    // Insert fragment at each unique cap position
    let lat_inv = structure.lattice.inverse_matrix();
    for anchor in &unique_anchors {
        let attachment_cart = lat_matrix * structure.frac_coords[anchor.node_attachment_atom];
        let aligned = align_fragment(fragment, &attachment_cart, &anchor.bond_direction);

        for (elem, cart_pos) in aligned.elements.iter().zip(aligned.cart_coords.iter()) {
            let frac = lat_inv * cart_pos;
            // Wrap into [0, 1)
            let wrapped = Vector3::new(
                frac.x - frac.x.floor(),
                frac.y - frac.y.floor(),
                frac.z - frac.z.floor(),
            );
            new_occupancies.push(SiteOccupancy {
                species: vec![(Species::neutral(*elem), 1.0)],
                properties: Default::default(),
            });
            new_frac_coords.push(wrapped);
        }
    }

    Ok(Structure {
        lattice: structure.lattice.clone(),
        site_occupancies: new_occupancies,
        frac_coords: new_frac_coords,
        pbc: structure.pbc,
        charge: structure.charge,
        properties: structure.properties.clone(),
    })
}

/// Align a molecular fragment so that:
/// - bonding_atom sits at `attachment_point + bond_direction * typical_bond_length`
/// - the fragment's internal orientation is rotated to align with bond_direction
fn align_fragment(
    fragment: &MolecularFragment,
    attachment_point: &Vector3<f64>,
    bond_direction: &Vector3<f64>,
) -> MolecularFragment {
    let bond_len = 1.5; // typical C-O or C-N bond length in Angstroms

    // Fragment's bonding atom position in its local frame
    let frag_origin = fragment.cart_coords[fragment.bonding_atom_idx];

    // Desired position: attachment_point + bond_direction * bond_len
    let target_pos = attachment_point + bond_direction * bond_len;

    // Compute fragment's "natural" bond direction from bonding atom
    // Use the centroid-to-bonding-atom vector, or default to -Z if only 1 atom
    let frag_dir = if fragment.cart_coords.len() > 1 {
        let centroid: Vector3<f64> = fragment.cart_coords.iter().sum::<Vector3<f64>>()
            / fragment.cart_coords.len() as f64;
        let dir = frag_origin - centroid;
        if dir.norm() > 1e-10 { dir.normalize() } else { -Vector3::z() }
    } else {
        -Vector3::z()
    };

    // Rotation matrix to align frag_dir with bond_direction
    let rotation = rotation_between(&frag_dir, bond_direction);

    // Transform all fragment atoms
    let new_coords: Vec<Vector3<f64>> = fragment.cart_coords.iter().map(|pos| {
        let centered = pos - frag_origin;
        let rotated = rotation * centered;
        rotated + target_pos
    }).collect();

    MolecularFragment {
        elements: fragment.elements.clone(),
        cart_coords: new_coords,
        bonding_atom_idx: fragment.bonding_atom_idx,
    }
}

/// Compute rotation matrix that rotates vector `from` to align with `to`.
/// Uses Rodrigues' rotation formula.
fn rotation_between(from: &Vector3<f64>, to: &Vector3<f64>) -> nalgebra::Matrix3<f64> {
    let f = from.normalize();
    let t = to.normalize();
    let dot = f.dot(&t);

    if dot > 0.9999 {
        return nalgebra::Matrix3::identity();
    }
    if dot < -0.9999 {
        // 180 degree rotation — find perpendicular axis
        let perp = if f.x.abs() < 0.9 { Vector3::x() } else { Vector3::y() };
        let axis = f.cross(&perp).normalize();
        return nalgebra::Matrix3::identity() - 2.0 * f * f.transpose()
            + 2.0 * (axis * axis.transpose() - nalgebra::Matrix3::identity())
            * (1.0 - dot) / (1.0 + dot + 1e-10); // approximate
    }

    let cross = f.cross(&t);
    let skew = nalgebra::Matrix3::new(
        0.0, -cross.z, cross.y,
        cross.z, 0.0, -cross.x,
        -cross.y, cross.x, 0.0,
    );
    nalgebra::Matrix3::identity() + skew + skew * skew / (1.0 + dot)
}

@router.post("/smiles-to-xyz")
async def smiles_to_xyz(data: dict):
    """Convert a SMILES string to 3D coordinates using RDKit or Open Babel."""
    smiles = data.get("smiles", "")
    if not smiles:
        raise HTTPException(400, "Missing SMILES string")

    try:
        from rdkit import Chem
        from rdkit.Chem import AllChem
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            raise ValueError(f"Invalid SMILES: {smiles}")
        mol = Chem.AddHs(mol)
        AllChem.EmbedMolecule(mol, AllChem.ETRFG_v3())
        AllChem.MMFFOptimizeMolecule(mol)

        elements = [atom.GetSymbol() for atom in mol.GetAtoms()]
        conf = mol.GetConformer()
        coords = [[conf.GetAtomPosition(i).x, conf.GetAtomPosition(i).y, conf.GetAtomPosition(i).z]
                   for i in range(mol.GetNumAtoms())]
        # bonding_atom_idx = 0 (first atom, typically the attachment point)
        # User can override in the UI later
        return {
            "elements": elements,
            "cart_coords": coords,
            "bonding_atom_idx": 0,
        }
    except ImportError:
        # Fallback to Open Babel
        import subprocess, json, tempfile
        with tempfile.NamedTemporaryFile(suffix=".smi", mode="w", delete=False) as f:
            f.write(smiles)
            smi_path = f.name
        xyz_path = smi_path.replace(".smi", ".xyz")
        subprocess.run(["obabel", smi_path, "-O", xyz_path, "--gen3d", "-h"], check=True)
        # Parse XYZ
        with open(xyz_path) as f:
            lines = f.readlines()
        n_atoms = int(lines[0].strip())
        elements = []
        coords = []
        for line in lines[2:2+n_atoms]:
            parts = line.split()
            elements.append(parts[0])
            coords.append([float(parts[1]), float(parts[2]), float(parts[3])])
        return {"elements": elements, "cart_coords": coords, "bonding_atom_idx": 0}
```
